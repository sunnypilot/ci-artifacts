41class TrainingGuide(Widget): 

42 def __init__(self, completed_callback=None): 

43 super().__init__() 

44 self._completed_callback = completed_callback

46 self._step = 0 

47 self._load_image_paths()

50 self._textures = [gui_app.texture(self._image_paths[0])]

53 threading.Thread(target=self._preload_thread, daemon=True).start()

55 def _load_image_paths(self): 

56 paths = [fn for fn in os.listdir(os.path.join(BASEDIR, "openpilot/selfdrive/assets/training")) if re.match(r'^step\d*\.png$', fn)] 

57 paths = sorted(paths, key=lambda x: int(re.search(r'\d+', x).group())) 

58 self._image_paths = [os.path.join(BASEDIR, "openpilot/selfdrive/assets/training", fn) for fn in paths] 

59 

60 def _preload_thread(self): 

61 # PNG loading is slow in raylib, so we preload in a thread and upload to GPU in main thread 

62 # We've already loaded the first image on init 

63 for path in self._image_paths[1:]: 

64 self._image_objs.append(gui_app._load_image_from_path(path)) 

65 

66 def _handle_mouse_release(self, mouse_pos): 

67 if rl.check_collision_point_rec(mouse_pos, STEP_RECTS[self._step]): 

68 # Record DM camera? 

69 if self._step == DM_RECORD_STEP: 

70 yes = rl.check_collision_point_rec(mouse_pos, DM_RECORD_YES_RECT) 

71 print(f"putting RecordFront to {yes}") 

72 ui_state.params.put_bool("RecordFront", yes, block=True) 

73 

74 # Restart training? 

75 elif self._step == len(self._image_paths) - 1: 

76 if rl.check_collision_point_rec(mouse_pos, RESTART_TRAINING_RECT): 

77 self._step = -1 

78 

79 self._step += 1 

80 

81 # Finished? 

82 if self._step >= len(self._image_paths): 

83 self._step = 0 

84 if self._completed_callback: 

85 self._completed_callback() 

86 

87 # NOTE: this pops OnboardingWindow during real onboarding 

88 gui_app.pop_widget() 

89 

90 def _update_state(self): 

91 if len(self._image_objs): 

92 self._textures.append(gui_app._load_texture_from_image(self._image_objs.pop(0))) 

93 

94 def _render(self, _): 

95 # Safeguard against fast tapping 

96 step = min(self._step, len(self._textures) - 1) 

97 rl.draw_texture_ex(self._textures[step], rl.Vector2(0, 0), 0.0, 1.0, rl.WHITE) 

98 

99 # progress bar 

100 if 0 < step < len(STEP_RECTS) - 1: 

101 h = 20 

102 w = int((step / (len(STEP_RECTS) - 1)) * self._rect.width) 

103 rl.draw_rectangle(int(self._rect.x), int(self._rect.y + self._rect.height - h), 

104 w, h, rl.Color(70, 91, 234, 255)) 

105 

106 if DEBUG: 

107 rl.draw_rectangle_lines_ex(STEP_RECTS[step], 3, rl.RED) 

108 

109 return -1
