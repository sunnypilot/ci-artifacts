117 def _update_calib_description(self): 

118 desc = tr(DESCRIPTIONS['reset_calibration']) 

119 

120 calib_bytes = self._params.get("CalibrationParams") 

121 if calib_bytes: 

122 try: 

123 calib = messaging.log_from_bytes(calib_bytes, log.Event).liveCalibration 

124 

125 if calib.calStatus != log.LiveCalibrationData.Status.uncalibrated: 

126 pitch = math.degrees(calib.rpyCalib[1]) 

127 yaw = math.degrees(calib.rpyCalib[2]) 

128 desc += tr(" Your device is pointed {:.1f}° {} and {:.1f}° {}.").format(abs(pitch), tr("down") if pitch > 0 else tr("up"), 

129 abs(yaw), tr("left") if yaw > 0 else tr("right")) 

130 except Exception: 

131 cloudlog.exception("invalid CalibrationParams") 

132 

133 lag_perc = 0 

134 lag_bytes = self._params.get("LiveDelay") 

135 if lag_bytes: 

136 try: 

137 lag_perc = messaging.log_from_bytes(lag_bytes, log.Event).liveDelay.calPerc 

138 except Exception: 

139 cloudlog.exception("invalid LiveDelay") 

140 if lag_perc < 100: 

141 desc += tr("<br><br>Steering lag calibration is {}% complete.").format(lag_perc) 

142 else: 

143 desc += tr("<br><br>Steering lag calibration is complete.") 

144 

145 torque_bytes = self._params.get("LiveTorqueParameters") 

146 if torque_bytes: 

147 try: 

148 torque = messaging.log_from_bytes(torque_bytes, log.Event).liveTorqueParameters 

149 # don't add for non-torque cars 

150 if torque.useParams: 

151 torque_perc = torque.calPerc 

152 if torque_perc < 100: 

153 desc += tr(" Steering torque response calibration is {}% complete.").format(torque_perc) 

154 else: 

155 desc += tr(" Steering torque response calibration is complete.") 

156 except Exception: 

157 cloudlog.exception("invalid LiveTorqueParameters") 

158 

159 desc += "<br><br>" 

160 desc += tr("sunnypilot is continuously calibrating, resetting is rarely required. " + 

161 "Resetting calibration will restart sunnypilot if the car is powered on.") 

162 

163 self._reset_calib_btn.set_description(desc)
