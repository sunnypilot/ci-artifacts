22def quantized_lru_cache(maxsize=128): 

23 def decorator(func): 

24 cache = OrderedDict() 

25 @wraps(func) 

26 def wrapper(cx, cy, r_mid, thickness, a0_deg, a1_deg, **kwargs): 

27 # Quantize inputs: balanced for smoothness vs cache effectiveness 

28 key = (round(cx), round(cy), round(r_mid), 

29 round(thickness), # 1px precision for smoother height transitions 

30 round(a0_deg * 10) / 10, # 0.1° precision for smoother angle transitions 

31 round(a1_deg * 10) / 10, 

32 tuple(sorted(kwargs.items()))) 

33 

34 if key in cache: 

35 cache.move_to_end(key) 

36 else: 

37 if len(cache) >= maxsize: 

38 cache.popitem(last=False) 

39 

40 result = func(cx, cy, r_mid, thickness, a0_deg, a1_deg, **kwargs) 

41 cache[key] = result 

42 return cache[key] 

43 return wrapper 

44 return decorator

55 def get_cap(left: bool, a_deg: float): 

56 # end cap at a1: center (a1), sweep a1→a1+180 (skip endpoints to avoid dupes) 

57 # quarter arc (outer corner) at a1 with fixed pixel radius cap_radius 

58 

59 nx, ny = math.cos(math.radians(a_deg)), math.sin(math.radians(a_deg)) # outward normal 

60 tx, ty = -ny, nx # tangent (CCW) 

61 

62 mx, my = cx + nx * r_mid, cy + ny * r_mid # mid-point at a1 

63 if DEBUG: 

64 rl.draw_circle(int(mx), int(my), 4, rl.PURPLE) 

65 

66 ex = mx + nx * (half - cap_radius) 

67 ey = my + ny * (half - cap_radius) 

68 

69 if DEBUG: 

70 rl.draw_circle(int(ex), int(ey), 2, rl.WHITE) 

71 

72 # sweep 90° in the local (t,n) frame: from outer edge toward inside 

73 if not left: 

74 alpha = np.deg2rad(np.linspace(90, 0, cap_segs + 2))[1:-1] 

75 else: 

76 alpha = np.deg2rad(np.linspace(180, 90, cap_segs + 2))[1:-1] 

77 cap_end = np.c_[ex + np.cos(alpha) * cap_radius * tx + np.sin(alpha) * cap_radius * nx, 

78 ey + np.cos(alpha) * cap_radius * ty + np.sin(alpha) * cap_radius * ny] 

79 

80 # bottom quarter (inner corner) at a1 

81 ex2 = mx + nx * (-half + cap_radius) 

82 ey2 = my + ny * (-half + cap_radius) 

83 if DEBUG: 

84 rl.draw_circle(int(ex2), int(ey2), 2, rl.WHITE) 

85 

86 if not left: 

87 alpha2 = np.deg2rad(np.linspace(0, -90, cap_segs + 1))[:-1] # include 0 once, exclude -90 

88 else: 

89 alpha2 = np.deg2rad(np.linspace(90 - 90 - 90, 0 - 90 - 90, cap_segs + 1))[:-1] 

90 cap_end_bot = np.c_[ex2 + np.cos(alpha2) * cap_radius * tx + np.sin(alpha2) * cap_radius * nx, 

91 ey2 + np.cos(alpha2) * cap_radius * ty + np.sin(alpha2) * cap_radius * ny] 

92 

93 # append to the top quarter 

94 if not left: 

95 cap_end = np.vstack((cap_end, cap_end_bot)) 

96 else: 

97 cap_end = np.vstack((cap_end_bot, cap_end)) 

98 

99 return cap_end 

100 

101 if a1_deg < a0_deg: 

102 a0_deg, a1_deg = a1_deg, a0_deg 

103 half = thickness * 0.5 

104 

105 cap_radius = min(cap_radius, half) 

106 

107 span = max(1e-3, a1_deg - a0_deg) 

108 

109 # pick arc segment count from arc length, clamp to shader points[] budget 

110 arc_len = r_mid * math.radians(span) 

111 arc_segs = max(6, int(arc_len / px_per_seg)) 

112 max_arc = (max_points - (4 * cap_segs + 3)) // 2 

113 arc_segs = max(6, min(arc_segs, max_arc)) 

114 

115 # outer arc a0→a1 

116 ang_o = np.deg2rad(np.linspace(a0_deg, a1_deg, arc_segs + 1)) 

117 outer = np.c_[cx + np.cos(ang_o) * (r_mid + half), 

118 cy + np.sin(ang_o) * (r_mid + half)] 

119 

120 # end cap at a1 

121 cap_end = get_cap(False, a1_deg) 

122 

123 # inner arc a1→a0 

124 ang_i = np.deg2rad(np.linspace(a1_deg, a0_deg, arc_segs + 1)) 

125 inner = np.c_[cx + np.cos(ang_i) * (r_mid - half), 

126 cy + np.sin(ang_i) * (r_mid - half)] 

127 

128 # start cap at a0 

129 cap_start = get_cap(True, a0_deg) 

130 

131 pts = np.vstack((outer, cap_end, inner, cap_start, outer[:1])).astype(np.float32) 

132 

133 # Rotate to start from middle of cap for proper triangulation 

134 pts = np.roll(pts, cap_segs, axis=0) 

135 

136 if DEBUG: 

137 n = len(pts) 

138 idx = int(time.monotonic() * 12) % max(1, n) # speed: 12 pts/sec 

139 for i, (x, y) in enumerate(pts): 

140 j = (i - idx) % n # rotate the gradient 

141 t = j / n 

142 color = rl.Color(255, int(255 * (1 - t)), int(255 * t), 255) 

143 rl.draw_circle(int(x), int(y), 2, color) 

144 

145 return pts
